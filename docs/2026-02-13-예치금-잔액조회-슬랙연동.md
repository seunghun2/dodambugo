### nginx 전체 설정
```nginx
server {
    listen 80;
    server_name _;

    # Health check
    location /health {
        return 200 'OK';
        add_header Content-Type text/plain;
    }

    # Innopay 송금 프록시
    location /proxy/transfer {
        if ($request_method != POST) {
            return 405;
        }
        proxy_pass https://acct.innopay.co.kr/AcctOutTransReq.acct;
        proxy_set_header Host acct.innopay.co.kr;
        proxy_set_header Content-Type application/json;
        proxy_ssl_server_name on;
        proxy_ssl_protocols TLSv1.2 TLSv1.3;
    }

    # Innopay 예치금 잔액조회 프록시
    location /proxy/balance {
        if ($request_method != POST) {
            return 405;
        }
        proxy_pass https://acct.innopay.co.kr/AcctOutRemainSearch.acct;
        proxy_set_header Host acct.innopay.co.kr;
        proxy_set_header Content-Type application/json;
        proxy_ssl_server_name on;
        proxy_ssl_protocols TLSv1.2 TLSv1.3;
    }

    # Innopay 송금결과 조회 프록시
    location /proxy/search {
        if ($request_method != POST) {
            return 405;
        }
        proxy_pass https://acct.innopay.co.kr/AcctOutTransSearch.acct;
        proxy_set_header Host acct.innopay.co.kr;
        proxy_set_header Content-Type application/json;
        proxy_ssl_server_name on;
        proxy_ssl_protocols TLSv1.2 TLSv1.3;
    }
}
```
